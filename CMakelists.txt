cmake_minimum_required(VERSION 3.28)

project(game LANGUAGES C CXX)
set(CMAKE_C_STANDARD 23)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(SDL2 "Include SDL2" OFF)
option(SDL3 "Include SDL3" ON)
option(GLFW "Include GLFW" OFF)
option(D3D11 "Include D3D11" OFF)
option(DXTK "Include DXTK" OFF)
option(GLAD "Include GLAD." OFF)
option(OPENGL "Include OPENGL." OFF)
option(MAGIC_ENUM "Include Magic Enum" OFF)
option(WEBGPU "Include WebGPU-Dawn" OFF)
option(GLM "Include GLM" OFF)
option(VULKAN "Include Vulkan" OFF)
option(VKBOOTSTRAP "Include VkBootstrap" OFF)
option(STBLIBS "Include Stb Libs" OFF)
option(VMA "Include VMA" OFF)
option(TINYOBJ "Include TinyOBJ" OFF)

file(GLOB_RECURSE game_src ${CMAKE_CURRENT_SOURCE_DIR}/src/**.cpp)

add_executable(game)

target_sources(game PRIVATE ${game_src})
target_compile_options(game PRIVATE -Wno-switch-default)
target_compile_features(game PRIVATE cxx_std_23)

if(TINYOBJ)
    add_subdirectory(deps/tinyobjloader)

    target_link_libraries(game PRIVATE tinyobjloader)
endif()

if(VMA)
    add_subdirectory(deps/VulkanMemoryAllocator)
    target_link_libraries(YourGameEngine PRIVATE GPUOpen::VulkanMemoryAllocator)
endif()

if(STBLIBS)
    set(stb_path ${CMAKE_CURRENT_SOURCE_DIR}/vendor/stb)

    add_library(stb_libs STATIC ${stb_path}/stb_image_impl.c ${stb_path}/stb_truetype_impl.c)

    target_include_directories(stb_libs PUBLIC ${stb_path}/)

    target_include_directories(game PRIVATE stb_libs)
endif()

if(VULKAN)
    find_package(Vulkan REQUIRED)
    target_link_libraries(game PRIVATE Vulkan::Vulkan)

    if(VKBOOTSTRAP)
        add_subdirectory(deps/vk-bootstrap)
        target_link_libraries(game PRIVATE vk-bootstrap::vk-bootstrap)
    endif()

    find_program(GLSLC glslc REQUIRED)
    set(SHADER_DIR ${CMAKE_CURRENT_SOURCE_DIR}/src/shaders)
    set(SPIRV_DIR ${CMAKE_CURRENT_SOURCE_DIR}/build/shaders)

    file(GLOB_RECURSE GLSLC_SHADERS ${SHADER_DIR}/*.vert ${SHADER_DIR}/*.frag)

    foreach(SHADER ${GLSLC_SHADERS})
        get_filename_component(FILE ${SHADER} NAME)
        set(OUTPUT_SHADER ${SPIRV_DIR}/${FILE}.spv)

        add_custom_command(
            OUTPUT ${OUTPUT_SHADER}
            COMMAND ${GLSLC} ${SHADER} -o ${OUTPUT_SHADER}
            DEPENDS ${SHADER}
            COMMENT "Compiling shader ${FILE}"
        )
        list(APPEND SPIRV_SHADERS ${OUTPUT_SHADER})
    endforeach()

    add_custom_target(shaders ALL DEPENDS ${SPIRV_SHADERS})
    add_dependencies(game shaders)
endif()

if(GLM)
    add_subdirectory(deps/glm)
    option(BUILD_SHARED_LIBS "Building GLM statically" ON)
    target_link_libraries(game PRIVATE glm::glm)
endif()

if(WEBGPU)
    add_subdirectory(deps/WebGPU)
    set(WEBGPU_LINK_TYPE STATIC)
    target_link_libraries(game PRIVATE webgpu)
endif()

if(DXTK)
    message("Including DXTK")
    option(BUILD_SHARED_LIBS "Building DXTK statically" ON)
    add_subdirectory(deps/DirectXTK deps/DirectXTK/build)
    target_link_libraries(game PUBLIC DirectXTK)
endif()

if(SDL3)
    message("Including SDL3.")
    option(SDL_STATIC "Enable/Disable static linking for SDL3" ON)
    option(SDL_SHARED "Enable/Disable shared linking for SDL3" OFF)
    set(sdl_path ${CMAKE_CURRENT_SOURCE_DIR}/deps/SDL)
    set(sdl_build_path ${sdl_path}/build)
    file(MAKE_DIRECTORY ${sdl_build_path})
    add_subdirectory(${sdl_path} ${sdl_build_path} EXCLUDE_FROM_ALL)
    target_link_libraries(game PRIVATE SDL3::SDL3-static)
endif()

if(GLAD)
    message("Including GLAD.")
    set(glad_path ${CMAKE_CURRENT_SOURCE_DIR}/deps/glad)

    # create the library with the source file in the add_library call
    add_library(glad STATIC ${glad_path}/src/glad.c)
    target_include_directories(glad PUBLIC ${glad_path}/include) # exported by the glad target
    target_include_directories(game PUBLIC ${glad_path}/include/) # exported by the glad target

    # link glad to the executable (game then sees the include paths transitively)
    target_link_libraries(game PRIVATE glad)
endif()

if(SDL2)
    message("Including SDL3.")

    option(SDL_STATIC "Enable/Disable static linking for SDL2" ON)
    option(SDL_SHARED "Enable/Disable shared linking for SDL2" OFF)
    set(sdl_path ${CMAKE_CURRENT_SOURCE_DIR}/deps/SDL)
    set(sdl_build_path ${sdl_path}/build)
    file(MAKE_DIRECTORY ${sdl_build_path})
    add_subdirectory(${sdl_path} ${sdl_build_path} EXCLUDE_FROM_ALL)
    target_link_libraries(game PRIVATE SDL2::SDL2main SDL2::SDL2-static)
endif()

if(GLFW)
    message("Including GLFW.")
    set(GLFW_LIBRARY_TYPE STATIC)
    set(glfw_path ${CMAKE_CURRENT_SOURCE_DIR}/deps/glfw)
    set(glfw_build_path ${glfw_path}/build)
    file(MAKE_DIRECTORY ${glfw_build_path})
    add_subdirectory(${glfw_path} ${glfw_build_path} EXCLUDE_FROM_ALL)
    target_link_libraries(game PRIVATE glfw)
endif()

if(D3D11)
    message("Including D3D11.")
    target_link_libraries(game PRIVATE d3d11.lib
        dxgi.lib d3dcompiler.lib winmm.lib dxguid.lib winmm.lib)
    add_custom_target(copy_shaders ALL
        COMMAND ${CMAKE_COMMAND} -E copy_directory
        ${CMAKE_CURRENT_SOURCE_DIR}/src/shaders
        ${CMAKE_CURRENT_SOURCE_DIR}/build/shaders
        COMMENT "Copying shaders to build directory"
    )
    add_dependencies(game copy_shaders)
endif()

if(OPENGL)
    message("Including OpenGL.")
    find_package(OpenGL REQUIRED)

    if(WIN32)
        target_link_libraries(game PRIVATE OpenGL::GL gdi32)
    else()
        find_package(Threads REQUIRED)
        find_package(X11 REQUIRED)
        target_link_libraries(game PRIVATE OpenGL::GL Threads::Threads
            ${X11_LIBRARIES})
    endif()
endif()

if(MAGIC_ENUM)
    message("Including Magic Enum")
    set(magic_enum_path ${CMAKE_CURRENT_SOURCE_DIR}/deps/magic_enum)
    file(MAKE_DIRECTORY ${magic_enum_path}/build)
    add_subdirectory(${magic_enum_path} ${magic_enum_path}/build EXCLUDE_FROM_ALL)

    target_link_libraries(game PRIVATE magic_enum)
endif()
