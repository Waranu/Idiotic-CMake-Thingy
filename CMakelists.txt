cmake_minimum_required(VERSION 3.21)

set(CMAKE_CXX_SCAN_FOR_MODULES OFF CACHE BOOL "Disable scanning for C++ module imports" FORCE)
project(game LANGUAGES C CXX)
set(CMAKE_C_STANDARD 23)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_EXTENSIONS ON)

option(SDL2 "Include SDL2" OFF)
option(SDL3 "Include SDL3" OFF)
option(GLFW "Include GLFW" OFF)
option(D3D11 "Include D3D11" OFF)
option(DXTK "Include DXTK" OFF)
option(GLAD "Include GLAD." OFF)
option(OPENGL "Include OPENGL." OFF)
option(MAGIC_ENUM "Include Magic Enum" OFF)

option(GLM "Include GLM" OFF)
option(VULKAN "Include Vulkan" OFF)
option(VKBOOTSTRAP "Include VkBootstrap" OFF)
option(STBLIBS "Include Stb Libs" OFF)
option(VMA "Include VMA" OFF)
option(SPDLOG "Include SPDLOG" OFF)
option(TINYOBJ "Include TinyOBJ" OFF)
option(LIBASSERT "Include Libassert" OFF)
option(IMGUI "Include Imgui" OFF)
option(SOKOL "Include Sokol" ON)

file(GLOB_RECURSE game_src ${CMAKE_CURRENT_SOURCE_DIR}/src/**.cpp)

add_executable(game)

target_sources(game PRIVATE ${game_src})
target_compile_options(game PRIVATE -Wno-switch-default)
target_compile_features(game PRIVATE cxx_std_23)
target_compile_features(game PRIVATE c_std_23)

if(IMGUI)
    message(STATUS "Including library: IMGUI")
    set(imgui_path ${CMAKE_CURRENT_SOURCE_DIR}/vendor/imgui)
    set(imgui_backend_path ${imgui_path}/backends)
    set(imgui_table_src ${imgui_path}/imgui_tables.cpp)
    set(imgui_sources
        ${imgui_path}/imgui.cpp
        ${imgui_path}/imgui_draw.cpp
        ${imgui_path}/imgui_demo.cpp
        ${imgui_path}/imgui_widgets.cpp
        ${imgui_table_src}
    )

    if(SDL2 AND NOT SOKOL)
        list(APPEND imgui_sources
            ${imgui_backend_path}/imgui_impl_sdl2.cpp
            ${imgui_backend_path}/imgui_impl_opengl3.cpp)
    endif()

    if(PREFIXED_RAYLIB)
        list(APPEND imgui_sources ${imgui_backend_path}/rlImGui.cpp)
    endif()

    add_library(imgui STATIC)
    target_include_directories(imgui PUBLIC ${imgui_path}/)
    target_include_directories(imgui PUBLIC ${imgui_backend_path}/)
    target_sources(imgui PRIVATE ${imgui_sources})

    if(SDL AND NOT SOKOL)
        target_include_directories(imgui PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/vendor/SDL/include)
        target_link_libraries(imgui PRIVATE SDL2::SDL2-static opengl32)
    elseif(PREFIXED_RAYLIB)
        target_link_libraries(imgui PRIVATE raylib)
    else()
    endif()

    target_link_libraries(game PRIVATE imgui opengl32)
endif()

if(VULKAN)
    find_package(Vulkan REQUIRED)
    target_include_directories(game PUBLIC Vulkan::Vulkan)
    target_link_libraries(game PRIVATE Vulkan::Vulkan)

    if(VKBOOTSTRAP)
        add_subdirectory(deps/vk-bootstrap)
        target_link_libraries(game PRIVATE vk-bootstrap::vk-bootstrap)
    endif()

    find_program(GLSLC glslc REQUIRED)
    set(SHADER_DIR ${CMAKE_CURRENT_SOURCE_DIR}/src/shaders)
    set(SPIRV_DIR ${CMAKE_CURRENT_SOURCE_DIR}/build/shaders)

    file(GLOB_RECURSE GLSLC_SHADERS ${SHADER_DIR}/*.vert ${SHADER_DIR}/*.frag)

    foreach(SHADER ${GLSLC_SHADERS})
        get_filename_component(FILE ${SHADER} NAME)
        set(OUTPUT_SHADER ${SPIRV_DIR}/${FILE}.spv)

        add_custom_command(
            OUTPUT ${OUTPUT_SHADER}
            COMMAND ${GLSLC} ${SHADER} -o ${OUTPUT_SHADER}
            DEPENDS ${SHADER}
            COMMENT "Compiling shader ${FILE}"
        )
        list(APPEND SPIRV_SHADERS ${OUTPUT_SHADER})
    endforeach()

    add_custom_target(shaders ALL DEPENDS ${SPIRV_SHADERS})
    add_dependencies(game shaders)
endif()

if(SOKOL)
    set(sokol_path ${CMAKE_CURRENT_SOURCE_DIR}/vendor/sokol)

    set(SOKOL_HEADERS
        ${sokol_path}/sokol_gfx.h
        ${sokol_path}/sokol_app.h
        ${sokol_path}/sokol_imgui.h
        ${sokol_path}/sokol_glue.h)

    add_library(sokol STATIC ${sokol_path}/sokol.cpp ${SOKOL_HEADERS})
    target_include_directories(sokol INTERFACE ${sokol_path})
    target_link_libraries(sokol PUBLIC imgui)
    target_link_libraries(game PRIVATE sokol)

    # Compile shaders from src/assets (not build/assets - glob runs at configure time)
    find_program(SOKOL_SHDC sokol_shdc REQUIRED)
    set(SHADER_SRC_DIR ${CMAKE_CURRENT_SOURCE_DIR}/src/assets)
    set(SHADER_OUT_DIR ${CMAKE_CURRENT_SOURCE_DIR}/src/assets)

    file(GLOB_RECURSE GLSL_SHADERS ${SHADER_SRC_DIR}/*.glsl)

    foreach(SHADER ${GLSL_SHADERS})
        get_filename_component(FILE ${SHADER} NAME)
        set(OUTPUT_SHADER ${SHADER_OUT_DIR}/${FILE}.h)
        add_custom_command(
            OUTPUT ${OUTPUT_SHADER}
            COMMAND ${SOKOL_SHDC} -i ${SHADER} -o ${OUTPUT_SHADER} -l glsl430:hlsl5 -f sokol
            DEPENDS ${SHADER}
            COMMENT "Compiling shader ${FILE}"
        )
        list(APPEND COMPILED_SHADERS ${OUTPUT_SHADER})
    endforeach()

    # Copy assets to build dir
    add_custom_target(copy_assets ALL
        COMMAND ${CMAKE_COMMAND} -E copy_directory
        ${CMAKE_CURRENT_SOURCE_DIR}/src/assets
        ${CMAKE_CURRENT_SOURCE_DIR}/build/assets
        COMMENT "Copying assets to build directory"
    )

    add_custom_target(shaders ALL DEPENDS ${COMPILED_SHADERS})
    add_dependencies(game shaders copy_assets)
endif()
